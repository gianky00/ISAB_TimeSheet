(() => {
    let info = "--- SYNCROJOB SCAN REPORT ---\n";
    info += "URL: " + window.location.href + "\n\n[INPUTS]\n";

    document.querySelectorAll('input, textarea').forEach(el => {
        info += `Name: ${el.name} | ID: ${el.id} | Visible: ${el.offsetParent !== null}\n`;
    });

    info += "\n[TRIGGERS]\n";

    document.querySelectorAll('.x-form-trigger').forEach(el => {
        info += `ID: ${el.id} | Class: ${el.className}\n`;
    });

    info += "\n[LIST ITEMS]\n";

    document.querySelectorAll('li.x-boundlist-item').forEach(el => {
        info += `Text: ${el.innerText.trim()} | ID: ${el.id}\n`;
    });

    console.log(info);
    copy(info);
})();
